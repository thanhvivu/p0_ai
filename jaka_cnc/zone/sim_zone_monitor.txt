# !!! SIMULATION ONLY !!! zone_monitor without DO writes; run as Sub-thread1 next to sim_test1_motion.
# ZONE_MONITOR - run as Sub-thread1 from the main program. No motion commands.
# DO56 = TCP past DoorC1 (odd CNC side), DO57 = past DoorC2 (even CNC side),
# DO58 = TCP within 50 mm of the Pick point along Y_user or beyond it.
# A DO is written only when its state changes, with a log line. Warning only, not safety.
# X_user / Y_user = TCP in usrfr_thanh; formula inside if() (no arithmetic in assignments).
# DoorC1 / DoorC2 X from the danger-boundary references in the main program.
zone_x_door1 = 679.362
zone_x_door2 = -365.358
# Margin past the Door plane: the robot standing at a Door is NOT counted inside the CNC.
zone_door_margin = 20.0
# Pick point TCP in usrfr_thanh = [683.786,1237.165,330.754]; limit = Y_pick - 50 mm.
zone_y_pick_limit = 1187.165
zone_odd_last = -1
zone_even_last = -1
zone_pick_last = -1
log_message(1,"ZONE MONITOR STARTED")
while(1):
    zone_tcp = get_atl_tcp_pose()
    zone_odd = 0
    if((0.7056980154 * (zone_tcp[0] - 345.833) + 0.7085118376 * (zone_tcp[1] + 489.782) + 0.0011344638 * (zone_tcp[2] + 136.233)) > zone_x_door1 + zone_door_margin):
        zone_odd = 1
    end
    zone_even = 0
    if((0.7056980154 * (zone_tcp[0] - 345.833) + 0.7085118376 * (zone_tcp[1] + 489.782) + 0.0011344638 * (zone_tcp[2] + 136.233)) < zone_x_door2 - zone_door_margin):
        zone_even = 1
    end
    zone_pick = 0
    if((-0.7085100607 * (zone_tcp[0] - 345.833) + 0.7056988446 * (zone_tcp[1] + 489.782) - 0.0016231544 * (zone_tcp[2] + 136.233)) > zone_y_pick_limit):
        zone_pick = 1
    end
    zone_chg = 1
    if(zone_odd == zone_odd_last):
        zone_chg = 0
    end
    if(zone_chg == 1):
        if(zone_odd == 1):
            sleep(0.01)
            log_message(1,"DO56 ODD CNC ZONE ON")
        else:
            sleep(0.01)
            log_message(1,"DO56 ODD CNC ZONE OFF")
        end
    end
    zone_odd_last = zone_odd
    zone_chg = 1
    if(zone_even == zone_even_last):
        zone_chg = 0
    end
    if(zone_chg == 1):
        if(zone_even == 1):
            sleep(0.01)
            log_message(1,"DO57 EVEN CNC ZONE ON")
        else:
            sleep(0.01)
            log_message(1,"DO57 EVEN CNC ZONE OFF")
        end
    end
    zone_even_last = zone_even
    zone_chg = 1
    if(zone_pick == zone_pick_last):
        zone_chg = 0
    end
    if(zone_chg == 1):
        if(zone_pick == 1):
            sleep(0.01)
            log_message(1,"DO58 PICK ZONE ON")
        else:
            sleep(0.01)
            log_message(1,"DO58 PICK ZONE OFF")
        end
    end
    zone_pick_last = zone_pick
    sleep(0.05)
end
