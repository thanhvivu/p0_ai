# ZONE_MONITOR - run as Sub-thread1 from the main program. No motion commands.
# X_user / Y_user = TCP in usrfr_thanh (formula inside if: no arithmetic in assignments).
# DO56 = robot past DoorC1 (odd CNC side), DO57 = past DoorC2 (even CNC side), DO58 = pick zone.
zone_x_door1 = 679.362
zone_x_door2 = -365.358
# TODO: Y_user of the Pick point minus 50 mm. DO58 stays OFF until zone_y_enable = 1.
zone_y_pick_limit = 0.0
zone_y_enable = 0
while(1):
    zone_tcp = get_atl_tcp_pose()
    if((0.7056980154 * (zone_tcp[0] - 345.833) + 0.7085118376 * (zone_tcp[1] + 489.782) + 0.0011344638 * (zone_tcp[2] + 136.233)) > zone_x_door1):
        set_digital_output(4,55,1,1)
    else:
        set_digital_output(4,55,0,1)
    end
    if((0.7056980154 * (zone_tcp[0] - 345.833) + 0.7085118376 * (zone_tcp[1] + 489.782) + 0.0011344638 * (zone_tcp[2] + 136.233)) < zone_x_door2):
        set_digital_output(4,56,1,1)
    else:
        set_digital_output(4,56,0,1)
    end
    zone_pick = 0
    if(zone_y_enable == 1):
        if((-0.7085100607 * (zone_tcp[0] - 345.833) + 0.7056988446 * (zone_tcp[1] + 489.782) - 0.0016231544 * (zone_tcp[2] + 136.233)) > zone_y_pick_limit):
            zone_pick = 1
        end
    end
    if(zone_pick == 1):
        set_digital_output(4,57,1,1)
    else:
        set_digital_output(4,57,0,1)
    end
    sleep(0.05)
end
