# ZONE_MONITOR - run as Sub-thread1 from the main program. No motion commands.
# DO56 = TCP past DoorC1 + margin (odd CNC side), DO57 = past DoorC2 + margin (even CNC side),
# DO58 = TCP within 50 mm of the Pick point along Y_user or beyond. Warning only, not safety.
# The robot runs with user frame usrfr_thanh (id 7) + tool_thanh (id 1): get_atl_tcp_pose()
# already returns X/Y in usrfr_thanh (verified 2026-11-19), so no conversion formula.
# Thresholds (usrfr_thanh, tool_thanh): DoorC1 X 679.362, DoorC2 X -365.358, Pick Y 1237.165 - 50.
zone_door_margin = 20.0
zone_x1 = (679.362 + zone_door_margin)
zone_x2 = (-365.358 - zone_door_margin)
zone_y_pick_limit = 1187.165
zone_odd_last = -1
zone_even_last = -1
zone_pick_last = -1
log_message(1,"ZONE MONITOR STARTED USER FRAME 7 MARGIN 20 PICK Y 1187")
while(1):
    zone_t = get_atl_tcp_pose()
    zone_xu = zone_t[0]
    zone_yu = zone_t[1]
    zone_odd = 0
    zone_even = 0
    zone_pick = 0
    if(zone_xu > zone_x1):
        zone_odd = 1
    end
    if(zone_xu < zone_x2):
        zone_even = 1
    end
    if(zone_yu > zone_y_pick_limit):
        zone_pick = 1
    end
    # Write a DO only when its state changes, with a log line.
    if(zone_odd == 1 && zone_odd_last != 1):
        set_digital_output(4,55,1,1)
        log_message(1,"DO56 ODD CNC ZONE ON")
    elif(zone_odd == 0 && zone_odd_last != 0):
        set_digital_output(4,55,0,1)
        log_message(1,"DO56 ODD CNC ZONE OFF")
    end
    zone_odd_last = zone_odd
    if(zone_even == 1 && zone_even_last != 1):
        set_digital_output(4,56,1,1)
        log_message(1,"DO57 EVEN CNC ZONE ON")
    elif(zone_even == 0 && zone_even_last != 0):
        set_digital_output(4,56,0,1)
        log_message(1,"DO57 EVEN CNC ZONE OFF")
    end
    zone_even_last = zone_even
    if(zone_pick == 1 && zone_pick_last != 1):
        set_digital_output(4,57,1,1)
        log_message(1,"DO58 PICK ZONE ON")
    elif(zone_pick == 0 && zone_pick_last != 0):
        set_digital_output(4,57,0,1)
        log_message(1,"DO58 PICK ZONE OFF")
    end
    zone_pick_last = zone_pick
    sleep(0.05)
end
